def sample(pair=None, trajectory=0, speed=5):
    return {"arms": [
        {"joints_deg": list(q), "velocities_deg_s": [0]*7,
         "ratios": [speed, speed], "trajectory_state": trajectory,
         "state": {"cur": 1, "err": 0}}
        for q in (pair or m.HOME)]}


class Tests(unittest.TestCase):
    def test_speed_option_preserves_calibration_and_separate_acceleration(self):
        names = ("SPEED", "ACCELERATION", "PLANNER_SPEED",
                 "PLANNER_ACCELERATION", "MAX_PLANNED_SPEED")
        with patch.multiple(m, **{name: getattr(m, name) for name in names}):
            m.configure_speed(5, 5)
            self.assertAlmostEqual(m.PLANNER_SPEED, .01)
            m.configure_speed(60, 5)
            self.assertEqual((m.SPEED, m.ACCELERATION), (60, 5))
            self.assertAlmostEqual(m.PLANNER_SPEED, .12)
            self.assertAlmostEqual(m.PLANNER_ACCELERATION, .01)
            self.assertAlmostEqual(m.MAX_PLANNED_SPEED, 21.7)
            robot = Mock()
            robot.runtime_set_speed_ratio.return_value = 0
            runner = m.Runner(robot)
            state = sample()
            for arm in state["arms"]:
                arm["ratios"] = [60, 5]
            runner.snapshot = Mock(return_value=state)
            with patch.object(m, "emit"):
                runner.set_speed()
            self.assertEqual([call.args for call in robot.runtime_set_speed_ratio.call_args_list],
                             [(0, 60, 5), (1, 60, 5)])
            robot.runtime_run_traj.assert_not_called()

    def test_invalid_speed_options_rejected_without_changing_configuration(self):
        before = (m.SPEED, m.ACCELERATION, m.PLANNER_SPEED, m.PLANNER_ACCELERATION)
        for invalid in (0, -5, 100.01, float("nan"), float("inf")):
            for values in ((invalid, 5), (60, invalid)):
                with self.assertRaises(ValueError):
                    m.configure_speed(*values)
                self.assertEqual(
                    (m.SPEED, m.ACCELERATION, m.PLANNER_SPEED, m.PLANNER_ACCELERATION), before)

    def test_windows_entry_passes_settings_to_preflight_and_execute(self):
        with patch.object(sys, "path", [str(Path(__file__).resolve().parents[1])]+sys.path):
            import run_dual_arm_large as entry

    def test_continuous_join_preserves_samples_without_intermediate_holds(self):
        plans, start = [], copy.deepcopy(m.HOME)
        for number, delta in enumerate((1, -1, 0), 1):
            target = copy.deepcopy(m.HOME)
            for q in target:
                q[4] = delta
            count = 81
            arrays = [(ctypes.c_double*(count*7))(*[
                a+(b-a)*i/(count-1)
                for i in range(count) for a,b in zip(start[arm],target[arm])])
                for arm in (0, 1)]
            plans.append({"number": number, "start": start, "target": target,
                          "arrays": arrays, "count": count, "native_point_counts": [count]*2})
            start = target
        with patch.object(m, "emit"):
            joined = m.join_plans(plans)
        self.assertEqual(joined["count"], 241+m.ENDPOINT_HOLD_POINTS)
        self.assertEqual(joined["waypoint_sample_indices"], [80, 160, 240])
        for arm in (0, 1):
            expected = list(plans[0]["arrays"][arm])
            for plan in plans[1:]:
                expected.extend(list(plan["arrays"][arm])[7:])
            expected.extend(m.HOME[arm]*m.ENDPOINT_HOLD_POINTS)
            self.assertEqual(list(joined["arrays"][arm]), expected)
        self.assertEqual(joined["target"], m.HOME)
        self.assertTrue(joined["continuous"])

    def test_paced_upload_uses_one_buffer_and_ordered_packets(self):
        robot = Mock()
        robot._lock = nullcontext()
        robot.dll.FX_L0_Arm0_Runtime_InitTraj.return_value = 0
        send = robot.dll.FX_L0_Arm0_Runtime_SetTraj
        send.return_value = 0
        runner = m.Runner(robot)
        runner.snapshot = Mock(side_effect=[
            sample(trajectory=2), sample(trajectory=2), sample(trajectory=1),
            sample(trajectory=1), sample(trajectory=1), sample(trajectory=1),
            sample(trajectory=2)])
        rows = [m.HOME[0][:4]+[i*.001, 0, 0] for i in range(123)]
        flat = [x for row in rows for x in row]
        raw = (ctypes.c_double*len(flat))(*flat)
        with patch.object(m, "emit"), patch.object(m.time, "sleep"):
            runner.upload_paced(0, raw, 123, 2)
        robot.dll.FX_L0_Arm0_Runtime_InitTraj.assert_called_once_with(0, 123)
        self.assertEqual([c.args[:3] for c in send.call_args_list],
                         [(0, 0, 50), (0, 1, 50), (0, 2, 23)])
        self.assertEqual([x for c in send.call_args_list for x in c.args[3]], flat)
        robot.runtime_run_traj.assert_not_called()
        robot.config_set_traj.assert_not_called()

    def test_continuous_loop_requires_observed_motion_before_home_completion(self):
        for did_run in (True, False):
            robot = Mock()
            robot.runtime_run_traj.return_value = 3
            robot.runtime_stop_traj.return_value = 3
            runner = m.Runner(robot)
            runner.upload_paced = Mock()
            states = [sample(), sample(trajectory=2)]
            if did_run:
                states += [sample(trajectory=3)]
            states += [sample()]*100
            runner.snapshot = Mock(side_effect=states)
            plan = {"start": m.HOME, "target": m.HOME, "arrays": [[], []],
                    "count": 300, "number": 1, "continuous": True}
            with patch.object(m, "emit"), patch.object(m.time, "sleep"), \
                    patch.object(m.time, "monotonic", side_effect=itertools.count(0, 2)):
                if did_run:
                    runner.execute_segment(plan)
                    robot.runtime_stop_traj.assert_not_called()
                else:
                    with self.assertRaises(TimeoutError):
                        runner.execute_segment(plan)
                    robot.runtime_stop_traj.assert_called_once_with(3)
            robot.runtime_run_traj.assert_called_once_with(3)
            self.assertEqual(runner.upload_paced.call_count, 2)
            robot.config_set_traj.assert_not_called()

    def test_upload_chunks_preserve_samples_and_dual_arm_alignment(self):
        for count in (5, 249, 250, 251, 252, 760, 899, 1038, 2500):
            arrays = [(ctypes.c_double*(count*7))(
                *[arm*100000+row*10+j for row in range(count) for j in range(7)])
                for arm in (0, 1)]
            plan = {"number": 1, "count": count, "arrays": arrays,
                    "start": [[float(a[j]) for j in range(7)] for a in arrays],
                    "target": [[float(a[(count-1)*7+j]) for j in range(7)] for a in arrays]}
            chunks = m.upload_chunks(plan)
            seen = set()
            for i, chunk in enumerate(chunks):
                self.assertTrue(5 <= chunk["count"] <= 250)
                self.assertEqual(chunk["chunk_count"], len(chunks))
                begin, end = chunk.get("source_range", [0, count])
                seen.update(range(begin, end))
                if i:
                    self.assertEqual(chunk["start"], chunks[i-1]["target"])
                for arm in (0, 1):
                    raw = chunk["arrays"][arm]
                    self.assertEqual(list(raw[:(end-begin)*7]), list(arrays[arm][begin*7:end*7]))
                    self.assertEqual(list(raw[-7:]), plan["target"][arm] if i == len(chunks)-1
                                     else list(arrays[arm][(end-1)*7:end*7]))
                    for pos in range((end-begin)*7, len(raw), 7):
                        self.assertEqual(list(raw[pos:pos+7]), chunk["target"][arm])
            self.assertEqual(seen, set(range(count)))
            self.assertEqual(chunks[-1]["target"], plan["target"])
