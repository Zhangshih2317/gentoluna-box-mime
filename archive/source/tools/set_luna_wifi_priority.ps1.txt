$ErrorActionPreference = 'Stop'
$logPath = Join-Path $PSScriptRoot '../evidence/luna-wifi-priority-result.txt'
try {
    $wifi = Get-NetAdapter -Name 'WLAN'
    if ($wifi.InterfaceGuid.ToString().Trim('{}') -ne 'b4b49fe9-983e-4b47-bf7e-bb8da09df1df') {
        throw 'WLAN adapter identity changed'
    }
    $before = Get-NetIPInterface -InterfaceIndex $wifi.ifIndex -AddressFamily IPv4
    $before | Select-Object InterfaceIndex,InterfaceAlias,AutomaticMetric,InterfaceMetric |
        ConvertTo-Json | Set-Content -LiteralPath $logPath -Encoding UTF8
    Set-NetIPInterface -InterfaceIndex $wifi.ifIndex -AddressFamily IPv4 `
        -AutomaticMetric Disabled -InterfaceMetric 500
    Get-NetIPInterface -InterfaceIndex $wifi.ifIndex -AddressFamily IPv4 |
        Select-Object InterfaceIndex,InterfaceAlias,AutomaticMetric,InterfaceMetric |
        ConvertTo-Json | Add-Content -LiteralPath $logPath -Encoding UTF8
    'SUCCESS' | Add-Content -LiteralPath $logPath -Encoding UTF8
} catch {
    $_ | Out-String | Add-Content -LiteralPath $logPath -Encoding UTF8
    exit 1
}
