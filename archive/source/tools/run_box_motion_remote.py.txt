"""Dispatch the empty-handed gesture and preserve its remote evidence."""

from datetime import datetime, timedelta, timezone
import importlib.util
import json
import sys
import uuid

from robot_status import DEFAULT_CONNECTOR, ROOT


def run(mode, squat_deg, turn_deg):
    if mode not in ("--preflight", "--execute"):
        raise ValueError("Invalid mode")
    if not 0 < squat_deg <= 30 or abs(turn_deg) != 45:
        raise ValueError("Invalid squat or waist angle")
    spec = importlib.util.spec_from_file_location("luna_connector", DEFAULT_CONNECTOR)
    connector = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(connector)
    stamp = datetime.now(timezone(timedelta(hours=8)))
    path = ROOT/"evidence"/("box-motion-"+stamp.strftime("%Y%m%d-%H%M%S")+".jsonl")
    path.parent.mkdir(exist_ok=True)
    token, pid = uuid.uuid4().hex, None
    plan_source = (ROOT/"tools/box_motion_plan.py").read_text(encoding="utf-8")
    robot_source = (ROOT/"tools/box_motion_robot.py").read_text(encoding="utf-8")
    source = (
        "import os,sys,json,types\n"
        "print(json.dumps({'event':'remote_process','pid':os.getpid(),"
        "'run_id':os.environ['LUNA_BOX_RUN']}),flush=True)\n"
        "planner=types.ModuleType('box_motion_plan')\n"
        "sys.modules['box_motion_plan']=planner\n"
        "exec("+repr(plan_source)+",planner.__dict__)\n"+robot_source
    )
    command = (
        "bash -lc 'source /home/nvidia/gento-test-baseline/gento-test-env.sh"
        " && exec env LUNA_BOX_RUN="+token+" python3 -u - "+mode
        +" --squat-deg "+repr(float(squat_deg))+" --turn-deg "+repr(float(turn_deg))+" 2>&1'"
    )
    print("Evidence: "+str(path), flush=True)
    with connector.connect() as client, path.open("w", encoding="utf-8") as log:
        stdin, stdout, stderr = client.exec_command(command, timeout=600)
        try:
            stdin.write(source.encode("utf-8"))
            stdin.flush()
            stdin.channel.shutdown_write()
            for line in stdout:
                log.write(line)
                log.flush()
                print(line, end="", flush=True)
                if line.startswith("{"):
                    event = json.loads(line)
                    if event.get("event") == "remote_process" and event.get("run_id") == token:
                        pid = int(event["pid"])
            code = stdout.channel.recv_exit_status()
            log.write(json.dumps({"remote_exit_code": code})+"\n")
            return code
        except BaseException:
            if pid is not None:
                stop_source = (
                    "import os,signal\n"
                    f"pid={pid!r}\n"
                    f"token={('LUNA_BOX_RUN='+token).encode()!r}\n"
                    "try:\n"
                    "    with open('/proc/%d/environ'%pid,'rb') as f: values=f.read().split(b'\\0')\n"
                    "    if token not in values: raise RuntimeError('Process token mismatch')\n"
                    "    os.kill(pid,signal.SIGTERM)\n"
                    "except FileNotFoundError: pass\n"
                )
                try:
                    with connector.connect() as stop_client:
                        si, so, se = stop_client.exec_command("python3 -", timeout=15)
                        si.write(stop_source.encode())
                        si.flush()
                        si.channel.shutdown_write()
                        so.read()
                        if so.channel.recv_exit_status():
                            raise RuntimeError(se.read().decode("utf-8", errors="replace"))
                    stdout.channel.settimeout(15)
                    for line in stdout:
                        log.write(line)
                        log.flush()
                        print(line, end="", flush=True)
                except Exception as exc:
                    print("STOP UNCONFIRMED; use physical emergency stop: "+str(exc), flush=True)
            raise
