"""Query the existing gateway's computational FK service; never send motion."""

import argparse
import importlib.util
import json
from pathlib import Path

from robot_status import DEFAULT_CONNECTOR, ROOT


REMOTE = r'''
import json
import sys
import uuid
import rclpy
from gento_unified_interfaces.srv import ForwardKinematics, ManageHardwareConnection
from rosidl_runtime_py.convert import message_to_ordereddict

def emit(kind, data):
    print(json.dumps({"kind": kind, "data": data}, ensure_ascii=False), flush=True)

emit("schema", {
    "request": ForwardKinematics.Request().get_fields_and_field_types(),
    "response": ForwardKinematics.Response().get_fields_and_field_types(),
})
if not JOBS:
    sys.exit(0)
rclpy.init()
node = rclpy.create_node("home_offline_fk_" + uuid.uuid4().hex[:8])
lease = node.create_client(ManageHardwareConnection, "/gento/hardware/manage_connection")
fk = node.create_client(ForwardKinematics, "/gento/arm/forward_kinematics")
owner = "home-fk-" + uuid.uuid4().hex
acquired = False
exit_code = 0

def call(client, request):
    if not client.wait_for_service(timeout_sec=3):
        raise RuntimeError("Service unavailable")
    future = client.call_async(request)
    rclpy.spin_until_future_complete(node, future, timeout_sec=15)
    if not future.done() or future.result() is None:
        raise RuntimeError("Service timed out")
    return future.result()

def manage(operation):
    request = ManageHardwareConnection.Request()
    request.operation = operation
    request.client_id = owner
    request.lease_seconds = 30.0
    request.stop_before_release = False
    result = call(lease, request)
    if not result.success:
        raise RuntimeError(result.message)
    return result

try:
    before = manage("inspect")
    if not before.connected:
        raise RuntimeError("Existing gateway must already be connected")
    if before.owner_client_id and before.expires_in > 0:
        raise RuntimeError("Another client owns the lease")
    manage("acquire")
    acquired = True
    for job in JOBS:
        manage("renew")
        request = ForwardKinematics.Request()
        request.arm = job["arm"]
        request.joints_deg = [float(v) for v in job["joints_deg"]]
        result = call(fk, request)
        emit("fk", {"input": job, "response": message_to_ordereddict(result)})
        if not result.success:
            raise RuntimeError(result.message)
except Exception as exc:
    emit("error", str(exc))
    exit_code = 1
finally:
    if acquired:
        try:
            emit("handoff", message_to_ordereddict(manage("handoff")))
        except Exception as exc:
            emit("error", str(exc))
            exit_code = 1
    node.destroy_node()
    rclpy.shutdown()
sys.exit(exit_code)
'''


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--jobs", type=Path)
    parser.add_argument("--output", type=Path, required=True)
    args = parser.parse_args()
    jobs = json.loads(args.jobs.read_text(encoding="utf-8")) if args.jobs else []
    spec = importlib.util.spec_from_file_location("luna_connector", DEFAULT_CONNECTOR)
    module = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(module)
    with module.connect() as client:
        stdin, stdout, stderr = client.exec_command(
            "bash -lc 'source /home/nvidia/gento-test-baseline/gento-test-env.sh"
            " && PYTHONIOENCODING=utf-8 python3 -'", timeout=90
        )
        stdin.write(("JOBS = " + repr(jobs) + "\n" + REMOTE).encode("utf-8"))
        stdin.flush()
        stdin.channel.shutdown_write()
        text = stdout.read().decode("utf-8", errors="replace")
        errors = stderr.read().decode("utf-8", errors="replace")
        code = stdout.channel.recv_exit_status()
    report = {
        "motion_commands_sent": 0, "exit_code": code,
        "events": [json.loads(s) for s in text.splitlines() if s.startswith("{")],
        "stderr": errors,
    }
    args.output.parent.mkdir(parents=True, exist_ok=True)
    args.output.write_text(json.dumps(report, ensure_ascii=False, indent=2), encoding="utf-8")
    print(json.dumps(report, ensure_ascii=True, indent=2))
    return code


if __name__ == "__main__":
    raise SystemExit(main())
