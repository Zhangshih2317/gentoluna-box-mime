param(
    [string]$OutputDirectory = (Join-Path $PSScriptRoot '..\assets\battery-audio')
)

Add-Type -AssemblyName System.Speech
$taskOutput = [System.IO.Path]::GetFullPath($OutputDirectory)
New-Item -ItemType Directory -Path $taskOutput -Force | Out-Null
$taskClips = @'
{
  "battery-1-low": "\u4e00\u53f7\u7535\u6c60\u7535\u91cf\u4e0d\u8db3\u3002\u8bf7\u53ca\u65f6\u5145\u7535\u3002",
  "battery-2-low": "\u4e8c\u53f7\u7535\u6c60\u7535\u91cf\u4e0d\u8db3\u3002\u8bf7\u53ca\u65f6\u5145\u7535\u3002",
  "battery-1-critical": "\u4e00\u53f7\u7535\u6c60\u7535\u91cf\u4e25\u91cd\u4e0d\u8db3\u3002\u8bf7\u5c3d\u5feb\u5145\u7535\u3002",
  "battery-2-critical": "\u4e8c\u53f7\u7535\u6c60\u7535\u91cf\u4e25\u91cd\u4e0d\u8db3\u3002\u8bf7\u5c3d\u5feb\u5145\u7535\u3002",
  "charging-started": "\u5df2\u5f00\u59cb\u5145\u7535\u3002",
  "charging-stopped": "\u5145\u7535\u5df2\u505c\u6b62\u3002"
}
'@ | ConvertFrom-Json
$taskSynth = New-Object System.Speech.Synthesis.SpeechSynthesizer
try {
    $taskSynth.SelectVoice('Microsoft Huihui Desktop')
    $taskSynth.Rate = -3
    $taskSynth.Volume = 90
    foreach ($taskClip in $taskClips.PSObject.Properties) {
        $taskSynth.SetOutputToWaveFile((Join-Path $taskOutput ($taskClip.Name + '.wav')))
        $taskSynth.Speak([string]$taskClip.Value)
        $taskSynth.SetOutputToNull()
    }
} finally {
    $taskSynth.Dispose()
}
Get-ChildItem -LiteralPath $taskOutput -Filter '*.wav' | Select-Object Name,Length
